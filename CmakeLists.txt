cmake_minimum_required(VERSION 3.16)

project(ICARUS LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# ---------------- Dependencies ----------------
find_package(CURL REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)

# ---------------- Core ----------------
add_library(core INTERFACE)

target_include_directories(core INTERFACE
    src
)

# ---------------- HTTP ----------------
add_library(http
    src/connectors/http/http_client.cpp
)

target_include_directories(http PUBLIC
    src
)

target_link_libraries(http PUBLIC
    CURL::libcurl
)

# ---------------- Kalshi ----------------
add_library(kalshi
    src/connectors/kalshi/kalshi_connector.cpp
    src/connectors/kalshi/kalshi_parsing.cpp
)

target_include_directories(kalshi PUBLIC
    src
)

target_link_libraries(kalshi PUBLIC
    core
    http
    nlohmann_json::nlohmann_json
)

# ---------------- Polymarket ----------------
add_library(polymarket
    src/connectors/polymarket/polymarket_connector.cpp
    src/connectors/polymarket/polymarket_parsing.cpp
)

target_include_directories(polymarket PUBLIC
    src
)

target_link_libraries(polymarket PUBLIC
    core
    http
    nlohmann_json::nlohmann_json
)

# ---------------- Connectors ----------------
add_library(connectors INTERFACE)

target_include_directories(connectors INTERFACE
    src
)

target_link_libraries(connectors INTERFACE
    kalshi
    polymarket
)

# ---------------- Canonical market catalog ----------------
add_library(market_catalog
    src/storage/market_catalog.cpp
)

target_include_directories(market_catalog PUBLIC
    src
)

target_link_libraries(market_catalog PUBLIC
    core
    nlohmann_json::nlohmann_json
)

# ---------------- Paper trading ----------------
add_library(paper_trading
    src/paper/paper_trading.cpp
)

target_include_directories(paper_trading PUBLIC
    src
)

target_link_libraries(paper_trading PUBLIC
    core
    nlohmann_json::nlohmann_json
)

# ---------------- App ----------------
add_library(app
    src/app/app.cpp
    src/app/engine.cpp
    src/storage/match_store.cpp
    src/strategy/convergence.cpp
    src/strategy/pricing.cpp
    src/strategy/detector.cpp
)

target_include_directories(app PUBLIC
    src
)

target_link_libraries(app PUBLIC
    connectors
    core
    market_catalog
    paper_trading
    nlohmann_json::nlohmann_json
)

# ---------------- Executable ----------------
add_executable(ICARUS
    src/main.cpp
)

target_link_libraries(ICARUS PRIVATE
    app
)

# ---------------- Tests ----------------
include(CTest)

if(BUILD_TESTING)
    add_executable(catalog_tests
        tests/catalog_tests.cpp
    )

    target_include_directories(catalog_tests PRIVATE
        src
    )

    target_link_libraries(catalog_tests PRIVATE
        kalshi
        polymarket
        market_catalog
    )

    add_test(NAME catalog_tests COMMAND catalog_tests)

    add_executable(match_store_tests
        tests/match_store_tests.cpp
        src/storage/match_store.cpp
    )

    target_include_directories(match_store_tests PRIVATE
        src
    )

    target_link_libraries(match_store_tests PRIVATE
        nlohmann_json::nlohmann_json
    )

    add_test(NAME match_store_tests COMMAND match_store_tests)

    add_executable(convergence_tests
        tests/convergence_tests.cpp
        src/strategy/convergence.cpp
        src/strategy/pricing.cpp
    )

    target_include_directories(convergence_tests PRIVATE
        src
    )

    target_link_libraries(convergence_tests PRIVATE
        core
    )

    add_test(NAME convergence_tests COMMAND convergence_tests)

    add_executable(paper_tests
        tests/paper_tests.cpp
    )

    target_include_directories(paper_tests PRIVATE
        src
    )

    target_link_libraries(paper_tests PRIVATE
        paper_trading
    )

    add_test(NAME paper_tests COMMAND paper_tests)
endif()

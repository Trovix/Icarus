cmake_minimum_required(VERSION 3.16)

project(ICARUS LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# ---------------- Dependencies ----------------
find_package(CURL REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)

# ---------------- Core ----------------
add_library(core INTERFACE)

target_include_directories(core INTERFACE
    src
)

# ---------------- HTTP ----------------
add_library(http
    src/connectors/http/http_client.cpp
)

target_include_directories(http PUBLIC
    src
)

target_link_libraries(http PUBLIC
    CURL::libcurl
)

# ---------------- Kalshi ----------------
add_library(kalshi
    src/connectors/kalshi/kalshi_connector.cpp
    src/connectors/kalshi/kalshi_parsing.cpp
)

target_include_directories(kalshi PUBLIC
    src
)

target_link_libraries(kalshi PUBLIC
    core
    http
    nlohmann_json::nlohmann_json
)

# ---------------- Polymarket ----------------
add_library(polymarket
    src/connectors/polymarket/polymarket_connector.cpp
    src/connectors/polymarket/polymarket_parsing.cpp
)

target_include_directories(polymarket PUBLIC
    src
)

target_link_libraries(polymarket PUBLIC
    core
    http
    nlohmann_json::nlohmann_json
)

# ---------------- Connectors ----------------
add_library(connectors INTERFACE)

target_include_directories(connectors INTERFACE
    src
)

target_link_libraries(connectors INTERFACE
    kalshi
    polymarket
)

# ---------------- App ----------------
add_library(app
    src/app/app.cpp
)

target_include_directories(app PUBLIC
    src
)

target_link_libraries(app PUBLIC
    connectors
    core
)

# ---------------- Executable ----------------
add_executable(ICARUS
    src/main.cpp
)

target_link_libraries(ICARUS PRIVATE
    app
    kalshi
)
